from .middleware import get_current_client

class ClientRouter:

    def db_for_read(self, model, **hints):

        # sirf contenttypes default me
        if model._meta.app_label in ['contenttypes', 'sessions']:
            return 'default'

        return get_current_client()

    def db_for_write(self, model, **hints):

        if model._meta.app_label in ['contenttypes', 'sessions']:
            return 'default'

        return get_current_client()

    def allow_relation(self, obj1, obj2, **hints):
        return True

    def allow_migrate(self, db, app_label, **hints):

        if app_label in ['contenttypes', 'sessions']:
            return db == 'default'

        # 🔥 auth + admin + your apps → client DB
        return True
